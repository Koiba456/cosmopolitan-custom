# A Window to Japan as Japan
country_event = {
	id = cosmo_history_events.4
	title = cosmo_history_events.4.t
	desc = cosmo_history_events.4.d
	picture = cosmogfx_NAGASAKI_1A_eventPicture
	goto = historyprovince

	is_triggered_only = yes
	
	trigger = {
		ai = no	
	}


	immediate = {
	hidden_effect = {
		random_known_country = {
			limit = {
				NOT = {
					culture_group = japanese_g
				}
				4182 = {
					trade_share = {
						country = PREV
						share = 1
					}					
				}
			}
			save_event_target_as = foreigntrader		
		}
		4182 = {
			save_event_target_as = historyprovince		
		}																
		}
	}

	option = {			# We will exploit this!
		name = cosmo_history_events.4.a
		4182 = {
			center_of_trade = 1					
		}
		add_isolationism = -1
		event_target:foreigntrader = {
			add_opinion = {
				who = ROOT
				modifier = opinion_traders_in_x
			}			
		}
		every_country = {
			limit = {
				culture_group = japanese_g
				isolationism = 3
			}
			add_opinion = {
				who = ROOT
				modifier = cosmo_foreigners_in_japan
			}			
		}		
	}
	option = {			# We will exploit this!
		name = cosmo_history_events.4.b
		4182 = {
			set_province_flag = no_foreign_traders	
		}
		event_target:foreigntrader = {
			add_opinion = {
				who = ROOT
				modifier = opinion_refused_traders_in_x
			}			
		}		
		add_isolationism = 1		
	}	
}
