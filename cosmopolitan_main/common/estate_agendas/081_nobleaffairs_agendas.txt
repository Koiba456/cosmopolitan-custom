estate_nobleaffairs_expand_into_x = {
	can_select = {
		has_country_flag = cosmo_puagenda_jwlaunch
	}
	selection_weight = {
		factor = 2
	}
	pre_effect = {
		random_subject_country = {
			limit = {
				has_country_flag = cosmo_wants_to_join_war
			}
			save_event_target_as = agendasubject			
		}
		random_country = {
			limit = {
				has_country_flag = cosmo_enemy_country
			}
			save_event_target_as = enemycountry			
		}		
		random_province = {
			limit = {
				has_province_flag = cosmo_agenda_target
			}
			save_event_target_as = agendaprovince
		}
	}
	immediate_effect = {
		event_target:agendaprovince = {
			set_province_flag = cosmo_agenda_province_active
			save_event_target_as = cosmo_agenda_province
		}
		event_target:agendasubject = {
			set_country_flag = cosmo_subject_country_claim_active
			save_event_target_as = cosmo_agendasubject
		}		
	}
	on_invalid = {
		random_subject_country = {
			limit = {
				has_country_flag = cosmo_subject_country_claim_active
			}
			add_trust = {
				who = ROOT
				value = 25
			}
			if = {
				limit = {
					liberty_desire = 25
				}
				add_liberty_desire = -15			
			}
		}		
	}
	provinces_to_highlight = {
		area_for_scope_province = {
			province_id = event_target:agendaprovince
		}
	}
	task_requirements = {
		event_target:agendaprovince = {
			owned_by = event_target:agendasubject
		}
	}
	fail_if = {
		event_target:agendaprovince = {
			NOT = {
			owned_by = event_target:agendasubject
			}
			owner = {
				is_at_war = no
			}
		}
	}
	task_completed_effect = {
		event_target:agendasubject = {
			add_trust = {
				who = ROOT
				value = 25
			}
			if = {
				limit = {
					liberty_desire = 25
				}
				add_liberty_desire = -15			
			}
		}
	}
    failing_effect = {
 		event_target:agendasubject = {
			add_trust = {
				who = ROOT
				value = -25
			}
		}
    }
}
