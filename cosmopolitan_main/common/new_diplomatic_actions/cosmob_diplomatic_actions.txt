# INSTRUCTIONS:
#
# condition				A diplomatic action can have any number of condition blocks, each with its own
#						tooltip, potential and allow section
#
# 	tooltip					Sets a custom text string similar to the hardcoded limits
# 							If no tooltip is scripted, the tooltip for the actual trigger will be shown
#							Note that the custom tooltip is only shown if the allow trigger is NOT met
#
# 	potential				Determines if the trigger is applicable or not
# 	allow					Determines if the action is valid or not
#
# effect				A diplomatic action can only have one effect block

# ROOT					actor
# FROM					target

cosmo_entertain_merchants = {
    category = economy           # Which diplomatic action list the action will appear in.
    
    alert_index = 6             # The alert to use for the action. Refer to static_actions for the indexes. Optional.
    alert_tooltip = COSMO_MILITARY_SUPPORT_ALERT_TT           # The localisation key to display in the alert tooltip. Optional.
    
    
	require_acceptance = yes	# Whether the action needs to be accepted by the target country for the accept effect to execute.
    
    # Determines when the action is visible
     is_visible = {
		has_estate_privilege = estate_madams_entertain_merchants		
		 any_province = {
			country_or_vassal_holds = ROOT
			has_province_modifier = cosmo_entertaining_merchants
			trade_share = {
				country = FROM
				share = 1
			}
		 }			 
    }
    
    # Determines when the action can be taken
    is_allowed = {
		NOT = {
		war_with = FROM
		}
		 NOT = {
			 trade_embargo_by = FROM 
		 }		
        # variable_arithmetic_trigger can be used here to use export_to_variable to extract and compare values such as ruler_age, etc
    }
    
    # Which effects to execute on acceptance
    on_accept = {
		set_country_flag = entertaining_merchant_country					
		FROM = {
			set_country_flag = entertain_merchant_country
			add_country_modifier = {
				name = entertain_merchant_country
				duration = 3650  
				desc = entertain_merchant_country_DESC
				hidden = no	
			}
		country_event = {#prostitute menus
			id = cosmo_brothel_events.614
			}			
		}
    }
    
    # Which effects to execute on decline (if declinable). Optional.
    on_decline = {
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = rejected_alliance
			}
		}
    }
    
    # Controls the AI's acceptance of the action, if acceptable. Optional.
   ai_acceptance = {
       # Add an AI factor entry

		
		
		add_entry = {
			name = DEVELOPMENT_OF_ROOT
			export_to_variable = {
				which = ai_value
				value = states_development
			}
			multiply_variable = {
			which = ai_value 
			value = 0.25 
			}
		}
	
		add_entry = {
			name = COSMO_PRESTIGE_OF_ROOT
			export_to_variable = {
				which = ai_value
				value = prestige
			}
			multiply_variable = { which = ai_value value = 1 }
		}

		add_entry = {
			name = OPINION
			export_to_variable = {
				which = ai_value
				value = opinion
				who = FROM
			}
			multiply_variable = { which = ai_value value = 1 }
		}

		add_entry = {
			name = COSMO_HAS_FRIENDLY_ATTITUDE
			limit = {
				FROM = {
					ai_attitude = { who = ROOT attitude = attitude_friendly }
				}
			}
			export_to_variable = {
				which = ai_value
				value = 50
			}
		}

		add_entry = {
			name = COSMO_TRUST
			
			export_to_variable = {
				variable_name = ai_value # Each "add_entry" must create one variable called "ai_value", the value of which will be extracted after executing all effects
