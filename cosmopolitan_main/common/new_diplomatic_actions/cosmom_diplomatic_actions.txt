#						tooltip, potential and allow section
#
# 	tooltip					Sets a custom text string similar to the hardcoded limits
# 							If no tooltip is scripted, the tooltip for the actual trigger will be shown
#							Note that the custom tooltip is only shown if the allow trigger is NOT met
#
# 	potential				Determines if the trigger is applicable or not
# 	allow					Determines if the action is valid or not
#
# effect				A diplomatic action can only have one effect block

# ROOT					actor
# FROM					target

cosmo_request_advisors = {
    category = alliance           # Which diplomatic action list the action will appear in.
    
    alert_index = 6             # The alert to use for the action. Refer to static_actions for the indexes. Optional.
    alert_tooltip = COSMO_REQUEST_ADVISORS_ALERT_TT           # The localisation key to display in the alert tooltip. Optional.
    
    
	require_acceptance = yes	# Whether the action needs to be accepted by the target country for the accept effect to execute.
    
    # Determines when the action is visible
    is_visible = {
		ai = no
    }
    
    # Determines when the action can be taken
    is_allowed = {
			prestige = 10
			FROM = {
			OR = {
				is_subject_of = ROOT
				alliance_with = ROOT
				overlord_of = ROOT
			}
		}
        # variable_arithmetic_trigger can be used here to use export_to_variable to extract and compare values such as ruler_age, etc
    }
    
    # Which effects to execute on acceptance
    on_accept = {
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = cosmo_requesting_advisors
			}
		}
		ROOT = {
			set_country_flag = request_advisor_spawn
		}	
    }
    
    # Which effects to execute on decline (if declinable). Optional.
    on_decline = {
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = rejected_alliance
			}
		}
    }
    
    # Controls the AI's acceptance of the action, if acceptable. Optional.
   ai_acceptance = {
       # Add an AI factor entry

		add_entry = {
			name = COSMO_HAS_PERSONAL_INTRODUCTION_OPINION
			limit = {
				FROM = {
					has_opinion_modifier = {
						who = ROOT
						modifier = cosmo_personal_introduction
					}
				}
			}
			export_to_variable = {
				which = ai_value
				value = 25
			}
			multiply_variable = { 
			which = ai_value 
			value = 1 }
		}
		
		add_entry = {
			name = DEVELOPMENT_OF_ROOT
			export_to_variable = {
				which = ai_value
				value = states_development
			}
			multiply_variable = {
			which = ai_value 
			value = 0.25 
			}
		}
		
		add_entry = {
			name = OPINION
			export_to_variable = {
				which = ai_value
				value = opinion
				who = FROM
			}
			multiply_variable = { which = ai_value value = 1 }
		}
	
		add_entry = {
			name = COSMO_PRESTIGE_OF_ROOT
			export_to_variable = {
				which = ai_value
				value = prestige
			}
			multiply_variable = { which = ai_value value = 0.25 }
		}
		
		add_entry = {
			name = COSMO_HAS_FRIENDLY_ATTITUDE
			limit = {
				FROM = {
					ai_attitude = { who = ROOT attitude = attitude_friendly }
				}
			}
			export_to_variable = {
				which = ai_value
				value = 50
			}
		}
		add_entry = {
			name = COSMO_HAS_THREATENED_ATTITUDE
			limit = {
				FROM = {
					ai_attitude = { who = ROOT attitude = attitude_threatened }
				}
			}
			export_to_variable = {
				which = ai_value
				value = 25
			}
		}
		add_entry = {
			name = COSMO_MILITARY_STRENGTH_ROOT

				which = ai_value
				value = army_size
			}

				which = ai_value

				}
		}
		
		add_entry = {
			name = COSMO_LOYAL_ALLY
			limit = {
				FROM = {
					AND = {
					alliance_with = ROOT

				}
			}

		}

				FROM = {
					AND = {
					is_subject_of = ROOT

					liberty_desire = 5

					}

				which = ai_value

		}

			name = COSMO_TRUST
